
class Solution {

    public int[] shortestAlternatingPaths(
        int n,
        int[][] redEdges,
        int[][] blueEdges
    ) {

        // Step 1: Create separate adjacency lists
        List<List<Integer>> red = new ArrayList<>();
        List<List<Integer>> blue = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            red.add(new ArrayList<>());
            blue.add(new ArrayList<>());
        }

        // Build red graph
        for (int[] edge : redEdges) {
            red.get(edge[0]).add(edge[1]);
        }

        // Build blue graph
        for (int[] edge : blueEdges) {
            blue.get(edge[0]).add(edge[1]);
        }

        // Step 2: Initialize result and visited
        int[] answer = new int[n];

        Arrays.fill(answer, -1);

        boolean[][] visited = new boolean[n][2];

        // Queue stores {node, lastColor}
        // 0 = red, 1 = blue
        Queue<int[]> queue = new LinkedList<>();

        // Start with both colors possible
        queue.offer(new int[]{0, 0});
        queue.offer(new int[]{0, 1});

        visited[0][0] = true;
        visited[0][1] = true;

        int distance = 0;

        // Step 3: BFS
        while (!queue.isEmpty()) {

            int size = queue.size();

            for (int i = 0; i < size; i++) {

                int[] curr = queue.poll();

                int node = curr[0];
                int lastColor = curr[1];

                // First time reaching this node
                if (answer[node] == -1) {
                    answer[node] = distance;
                }

                // If last edge was red, next must be blue
                if (lastColor == 0) {

                    for (int neighbor : blue.get(node)) {

                        if (!visited[neighbor][1]) {

                            visited[neighbor][1] = true;

                            queue.offer(
                                new int[]{neighbor, 1}
                            );
                        }
                    }
                }

                // If last edge was blue, next must be red
                else {

                    for (int neighbor : red.get(node)) {

                        if (!visited[neighbor][0]) {

                            visited[neighbor][0] = true;

                            queue.offer(
                                new int[]{neighbor, 0}
                            );
                        }
                    }
                }
            }

            distance++;
        }

        return answer;
    }
}
