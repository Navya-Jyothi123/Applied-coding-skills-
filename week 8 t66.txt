
class Solution {

    public int[][] floodFill(
        int[][] image,
        int sr,
        int sc,
        int color
    ) {

        int originalColor = image[sr][sc];

        // Important: Avoid infinite recursion
        if (originalColor == color) {
            return image;
        }

        dfs(image, sr, sc, originalColor, color);

        return image;
    }

    private void dfs(
        int[][] image,
        int row,
        int col,
        int originalColor,
        int color
    ) {

        int m = image.length;
        int n = image[0].length;

        // Check boundaries and original color
        if (row < 0 || row >= m ||
            col < 0 || col >= n ||
            image[row][col] != originalColor) {

            return;
        }

        // Change current pixel color
        image[row][col] = color;

        // Explore four directions
        dfs(image, row - 1, col, originalColor, color); // Up
        dfs(image, row + 1, col, originalColor, color); // Down
        dfs(image, row, col - 1, originalColor, color); // Left
        dfs(image, row, col + 1, originalColor, color); // Right
    }
}
