class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        int totalSum = 0;

        for (int num : nums) {
            totalSum += num;
        }

        int leftSum = 0;

        for (int i = 0; i < n; i++) {

            int leftCount = i;
            int rightCount = n - i - 1;

            int leftCost = nums[i] * leftCount - leftSum;

            int rightSum = totalSum - leftSum - nums[i];
            int rightCost = rightSum - nums[i] * rightCount;

            result[i] = leftCost + rightCost;

            leftSum += nums[i];
        }

        return result;
    }
}
Input
nums =[2,3,5]
Output
[4,3,5]
