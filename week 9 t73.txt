
import java.util.*;

class Solution {

    public String longestNiceSubstring(String s) {

        if (s.length() < 2) {
            return "";
        }

        HashSet<Character> set = new HashSet<>();

        // Store all characters
        for (char ch : s.toCharArray()) {
            set.add(ch);
        }

        // Find a character without its opposite case
        for (int i = 0; i < s.length(); i++) {

            char ch = s.charAt(i);

            if (Character.isLowerCase(ch)) {

                if (!set.contains(Character.toUpperCase(ch))) {

                    String left = longestNiceSubstring(
                        s.substring(0, i)
                    );

                    String right = longestNiceSubstring(
                        s.substring(i + 1)
                    );

                    return left.length() >= right.length()
                            ? left : right;
                }

            } else {

                if (!set.contains(Character.toLowerCase(ch))) {

                    String left = longestNiceSubstring(
                        s.substring(0, i)
                    );

                    String right = longestNiceSubstring(
                        s.substring(i + 1)
                    );

                    return left.length() >= right.length()
                            ? left : right;
                }
            }
        }

        // Every character has both cases
        return s;
    }
}
