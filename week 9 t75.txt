
import java.util.*;

class Solution {

    public int[][] kClosest(int[][] points, int k) {

        PriorityQueue<int[]> pq = new PriorityQueue<>(
            (a, b) -> Integer.compare(
                a[0] * a[0] + a[1] * a[1],
                b[0] * b[0] + b[1] * b[1]
            )
        );

        // Add all points to Min Heap
        for (int[] point : points) {
            pq.offer(point);
        }

        int[][] result = new int[k][2];

        // Extract k closest points
        for (int i = 0; i < k; i++) {
            result[i] = pq.poll();
        }

        return result;
    }
}
