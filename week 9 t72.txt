
class Solution {

    public int reverseBits(int n) {

        int result = 0;

        for (int i = 0; i < 32; i++) {

            // Shift result left
            result = result << 1;

            // Extract last bit and add it
            result = result | (n & 1);

            // Unsigned right shift
            n = n >>> 1;
        }

        return result;
    }
}
