
class Solution {
    public int findCircleNum(int[][] isConnected) {

        int n = isConnected.length;

        boolean[] visited = new boolean[n];

        int provinces = 0;

        // Visit every city
        for (int i = 0; i < n; i++) {

            if (!visited[i]) {

                provinces++;

                dfs(i, isConnected, visited);
            }
        }

        return provinces;
    }

    private void dfs(
        int city,
        int[][] isConnected,
        boolean[] visited
    ) {

        visited[city] = true;

        // Check all neighboring cities
        for (int neighbor = 0;
             neighbor < isConnected.length;
             neighbor++) {

            if (isConnected[city][neighbor] == 1 &&
                !visited[neighbor]) {

                dfs(neighbor, isConnected, visited);
            }
        }
    }
}
