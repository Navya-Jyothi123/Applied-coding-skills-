
class Solution {

    public int rob(int[] nums) {

        int n = nums.length;

        if (n == 1) {
            return nums[0];
        }

        int prev2 = 0;
        int prev1 = 0;

        for (int i = 0; i < n; i++) {

            // Rob current house
            int rob = nums[i] + prev2;

            // Skip current house
            int skip = prev1;

            int current = Math.max(rob, skip);

            prev2 = prev1;
            prev1 = current;
        }

        return prev1;
    }
}
