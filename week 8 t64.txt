
class Solution {

    public int[][] updateMatrix(int[][] mat) {

        int m = mat.length;
        int n = mat[0].length;

        int[][] dist = new int[m][n];

        Queue<int[]> queue = new LinkedList<>();

        // Step 1: Initialize all distances
        // Add all zero cells to the queue
        for (int i = 0; i < m; i++) {

            for (int j = 0; j < n; j++) {

                if (mat[i][j] == 0) {

                    dist[i][j] = 0;

                    queue.offer(new int[]{i, j});

                } else {

                    dist[i][j] = -1;
                }
            }
        }

        // Four directions: up, down, left, right
        int[][] directions = {
            {-1, 0},
            {1, 0},
            {0, -1},
            {0, 1}
        };

        // Step 2: Multi-source BFS
        while (!queue.isEmpty()) {

            int[] curr = queue.poll();

            int row = curr[0];
            int col = curr[1];

            // Explore all four directions
            for (int[] dir : directions) {

                int nr = row + dir[0];
                int nc = col + dir[1];

                // Check boundaries and visited status
                if (nr >= 0 && nr < m &&
                    nc >= 0 && nc < n &&
                    dist[nr][nc] == -1) {

                    dist[nr][nc] = dist[row][col] + 1;

                    queue.offer(new int[]{nr, nc});
                }
            }
        }

        return dist;
    }
}
